180. Consecutive Numbers

--------------------------------------------------------------------
Solution 1 =>

select distinct l1.num as ConsecutiveNums
from logs l1, logs l2, logs l3
where l1.id = l2.id + 1
    and l2.id = l3.id + 1
    and l1.num = l2.num
    and l2.num = l3.num

--------------------------------------------------------------------
Solution 2 => 
 
select distinct logs1.num as ConsecutiveNums from logs as logs1
join logs as logs2 on logs1.id + 1 = logs2.id
join logs as logs3 on logs2.id + 1 = logs3.id
where logs1.num = logs2.num and logs2.num = logs3.num

--------------------------------------------------------------------
Solution 3 => 

select distinct num as ConsecutiveNums
from (
    select
        num,
        lag(num) over (order by id) as prev_num,
        lead(num) over (order by id) as next_num
    from logs
) as subquery
where num = prev_num and num = next_num

--------------------------------------------------------------------
Solution 4 (Generic, for N number of nums) =>

WITH NumberGroups AS (
    SELECT 
        num,
        id,
        id - ROW_NUMBER() OVER (PARTITION BY num ORDER BY id) AS grp
    FROM 
        Logs
),
ConsecutiveCounts AS (
    SELECT 
        num,
        COUNT(*) AS consecutive_count
    FROM 
        NumberGroups
    GROUP BY 
        num, grp
)
SELECT DISTINCT num AS ConsecutiveNums
FROM 
    ConsecutiveCounts
WHERE 
    consecutive_count >= 3;

Объяснение:

NumberGroups CTE:
    ROW_NUMBER() OVER (PARTITION BY num ORDER BY id) нумерует строки внутри каждой группы с одинаковыми значениями num в порядке возрастания id.
    Выражение id - ROW_NUMBER() OVER (PARTITION BY num ORDER BY id) дает одинаковое значение grp для последовательных строк с одинаковым значением num, если id идут подряд.

ConsecutiveCounts CTE:
    Группируем записи по num и grp и считаем количество строк в каждой группе.

Заключительный SELECT:
    Выбираем только те числа, которые встречаются подряд три и более раз.
