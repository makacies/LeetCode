184. Department Highest Salary

WITH CTE AS (
    SELECT 
        Department.name AS Department, 
        Employee.name AS Employee, 
        salary, 
        RANK() OVER (PARTITION BY Department.name ORDER BY salary DESC) AS rankNum
    FROM 
        Employee
    JOIN 
        Department 
    ON 
        Employee.departmentId = Department.id
)
SELECT 
    Department, 
    Employee, 
    salary
FROM 
    CTE
WHERE 
    rankNum = 1;
